١٥٧٢ ـ إبراهيم بن مرسي الإبناسي		١٦٠٦ ـ إبراهيم بن علي السمنودي
أخــذ عــنه :
١٦٢٠ ـ عبدالحكيم بن عبدالسلام خاطر
أفادني عن سيرته بنفسه ـ رحمه الله تعالى ـ وعندي شريط كاسيت بذلك،
وأخذت تاريخ وفاته من ابنه الأستاذ أسامة، وقد ارتفع إلى هذه الحلقة لعلو السند
تلميذه الشيخ عبدالحكيم من طريق آخر .
١٦١٥ ـ  عَبْدُالعَزِيزِ بنُ عَبْدِاللهِ بنِ حَسَنِ بنِ حُسَيْنِ بنِ عَلِيِّ بنِ حُسَيْنِ ابنِ
الشَّيْخِ مُحَمَّدِ بنِ عَبْدِالوَهَّابِ النَّجْدِيّ الحَنْبَلِي إِمَامُ المَسْجِدِ
الحَرَامِ وَوَزِيرُ المَعَارِفِ السُّعُودِيُّ الأسْبَقُ
ولــد في: ١٣٣٦هـ.
توفي في: ١٤٠٤هـ.
أخــذ عــن:
١٥٥٩ ـ عبدالله بن حسن بن حسين النجدي
أفادني عن جوانب من ترجمته فضيلة الشيخ عبدالعزيز بن عبدالمحسن بن
عبدالله آل الشيخ، مدير التعليم ورئيس هيئة الأمر بالمعروف والنهي عن المنكر
بمحافظة بيشة بالمملكة العربية السعودية سابقًا.
١٦١٦ ـ  مَحْمُودُ بنُ أَمِينِ بنِ طَنْطَاوِي بنِ بَدَوِيِّ الشَّرْقَاوِيُّ ثُمَّ القَاهِرِيُّ
ولــد في: ١٠ رجب ١٣٤٩هـ.
١٦١٥ ـ علماء نجد خلال ثمانية قرون ٣/ ٤٣٣، أئمة المسجد الحرام ٣١.
