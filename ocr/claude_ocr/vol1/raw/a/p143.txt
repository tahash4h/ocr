١٥١٠ ـ محمَّد سليم اللبني                    ١٥٢٣ ـ عبدالقادر العربيلي
١٥٢٨ ـ ياسين بن صالح الجويجاني
أخـذ عـنه :
١٦٠٧ ـ صالح بن إدريس الأركاني
مـن أعمـالـه:
١ ـ «إتحاف حرز الأماني برواية الإصبهاني» .
٢ ـ «رسالة البيان في رسم القرآن» .
١٥٦١ ـ مُحَمَّدُ بنُ مُحَمَّدِ بنِ أَحْمَدَ العنانيُّ الطَّنْطَاوِيُّ المِصْرِيُّ
ولـد في: ١٣٥٧هـ تقريبًا .
أخـذ عـن:
١٤٧٩ ـ حسن الجريسي الكبير
جاء في أسانيد شيخنا الشيخ/ عبدالرَّازق السيد أحمد البكري وغيره، ولم أقف
علىٰ ترجمة تلميذه إبراهيم سعد الذي جاء في الأسانيد أنَّه أخذ عنه، فلعلَّنا نقف عليه
فيما بعد لامتداد سند هـٰذا الشيخ، رحم الله الجميع .
١٥٦٢ ـ مُحَمَّدُ بن مُحَمَّدِ بنِ عُمَرَ بنِ عَلِي بنِ سَالِم بنِ قَاسِم مَخْلُوف
الشَّرِيفُ التُّونِسِيُّ
ولـد في: ١٢٨٠هـ تقريبًا .
توفي في: ١٣٦٠هـ .

١٥٦٢ ـ فهرس الفهارس ١/ ٢٠٦، أعلام الزركلي ٧/ ٨٢، شجرة النور ٤٤٦، ٤٦١ .

---NOTES---
The name «عبدالرَّازق» is spelled this way on the page (not «عبدالرزاق»), and I kept it as printed.
