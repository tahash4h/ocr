١٤٨٨ ـ محمَّد سعيدالحلواني		١٥١١ ـ عزالدين العرقسوسي الدمشقي
١٥٤٥ ـ محمود بن فائز الدير عطاني		١٥٤٧ ـ عبدالعزيز عيون السود الحمصي
أخذ عنه:
١٦٣١ ـ أيمن بن سويد الدمشقي		١٦٣٤ ـ محمود تميم الزعبي
أفدناه من تلميذه فضيلة الأستاذ الدكتور/ أيمن سويد الدمشقي، وهـٰذا إلىٰ
جانب المرجع المذكور في الهامش، وترجمة أخرىٰ وصلتني عن طريق الشيخ/ سعيد
سعد الدربي الغامدي، بناء علىٰ طلبي أثناء رحلته إلىٰ دمشق في صيف ١٤٢٢هـ.
وقد أخذ عن الشيخ/ محيي الدين الكردي جمع كبير من القرَّاء لم تصلني عنهم
ترجمة وافية حتىٰ الآن، بارك الله في عمر الشيخ ونفع به الإسلام والمسلمين.
١٥٦٩ ـ مُصْطَفَىٰ بنُ مَنْصُورِ البَاجُورِيُّ المَنُوفِيُّ المِصْرِيُّ
توفي في: ١٣٧٤هـ تقريبًا.
أخذ عن:
١٤٧٣ ـ محمَّد بن مكي بن نصر.		١٥١٨ ـ علي بن سبيع القاهري
أخذ عنه:
١٦١٠ ـ عبدالحكيم بن عبداللَّطيف الخطيب
أفدناه من تلميذه وشيخنا الشَّيخ عبدالحكيم عبداللطيف، ويرتفع الشيخ
مصطفىٰ الباجوري درجة بأخذه عن محمَّد مكي نصر.
