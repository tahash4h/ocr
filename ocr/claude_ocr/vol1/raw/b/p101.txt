١٦٢٠ ـ عَبْدُالحَكِيمِ بنُ عَبْدِالسَّلامِ بنِ عَبْدِالحَفِيظِ خَاطِرِ الجِيزِيُّ المِصْرِيُّ
نزيلُ المدينةِ المنورةِ
ولــد في: ٢٢ شعبان ١٣٦٤هـ.
أخــذ عــن:
١٥٦٥ ـ أحمد بن عبدالعزيز الزيات	١٥٩٨ ـ عامر بن السيد بن عثمان
١٦٠٠ ـ أحمد بن علي بن مرعي	١٦٠٣ ـ عبدالفتاح عجمي المرصفي
١٦٠٨ ـ رزق بن خليل حبة	١٦١٤ ـ محمود برانق القليوبي
أفادني عن سيرته بنفسه، بارك الله في عمره وعمله.
١٦٢١ ـ مُحَمَّدُ بنُ حُسَيْنِ بنِ عَامِرِ الحَدائِيُّ الصَّنْعَانِيُّ اليَمَنِيُّ
ولــد في: ١٣٥٥هـ تقريبًا.
توفي في: ١٥ رمضان ١٤١٩هـ.
أخــذ عــن:
١٥٤٢ ـ علي بن عبدالله الجمالي الطائفي
أفادني بجوانب من ترجمة هذا الشيخ تلميذه فضيلة الشيخ يحيى أحمد الحليلي
مدرس القراءات بالكلية العليا للقرآن الكريم بصنعاء، وأخبرني أن أخذه عن الشيخ
علي بن عبدالله الطائفي كان في رواية حفص وهو صغير، وأن له طريق آخر في
القراءات وهو أخذه عن: حسن لطف باصيد عن: محمد علي الأكوع عن: حسين
مبارك الغيثي عن: أحمد ناصر الخولاني عن: علي أحمد السدمي عن: علي أحمد
الشرفي عن: ١٤٣٩ ـ يحيى بن هادي الشرقي، وبهذا يتصل سنده بالسلسلة، ولكن
هذا الطريق ينزل درجتين عن طريق الشيخ علي بن عبدالله الطائفي، وقد اشترك الشيخ
الحليلي مع شيخه في الأخذ عن باصيد.
---NOTES---
- The two-column entries are separated by a tab: right column first, then left column.
- "خَاطِرِ الجِيزِيُّ": the kasra under خاطر is hard to make out, and it may be a different vowel.
- "وأن له طريق آخر": this matches the print. Standard grammar would require "طريقًا", but I did not change it.
- "الحَدائِيُّ": the tashkeel on this name is my best reading of the scan.
